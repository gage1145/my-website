// Only the selected roll/keep pair is plotted.
subset = transpose(counts).filter((d) => +d.roll === roll && +d.keep === keep)
stats = {
  const rows = subset.slice().sort((a, b) => a.value - b.value);
  const n = d3.sum(rows, (d) => d.n);
  const hits = d3.sum(rows.filter((d) => d.value >= target), (d) => d.n);

  // Weighted median: walk the tallies until half the rolls are behind us.
  let seen = 0;
  let median = NaN;
  for (const row of rows) {
    seen += row.n;
    if (seen >= n / 2) {
      median = row.value;
      break;
    }
  }

  // Each row is a tally, so mean and SD must be weighted by n. d3.deviation
  // would treat every distinct total as a single roll and overweight the tails.
  const mean = n ? d3.sum(rows, (d) => d.value * d.n) / n : NaN;
  const sd = n > 1
    ? Math.sqrt(d3.sum(rows, (d) => d.n * (d.value - mean) ** 2) / (n - 1))
    : NaN;

  return {
    n: n,
    hits: hits,
    p: n ? hits / n : 0,
    mean: mean,
    sd: sd,
    median: median,
    min: rows.length ? rows[0].value : NaN,
    max: rows.length ? rows[rows.length - 1].value : NaN
  };
}
// Bin edges are anchored on the target so no bar straddles it: every bar is
// wholly below the target or wholly at/above it, which is what the fill colour
// claims.
bins = {
  const width = Math.max(1, Math.ceil((stats.max - stats.min + 1) / 36));
  const start = target - Math.ceil((target - stats.min) / width) * width;
  return {
    width: width,
    thresholds: d3.range(start, stats.max + width + 1, width)
  };
}
